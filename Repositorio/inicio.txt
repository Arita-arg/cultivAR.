<script>
        // Simple Interaction Logic
        document.querySelectorAll('input[type="checkbox"]').forEach(toggle => {
            toggle.addEventListener('change', (e) => {
                const statusText = e.target.closest('.neomorphic-card').querySelector('p');
                if(e.target.checked) {
                    statusText.classList.remove('text-on-surface-variant');
                    statusText.classList.add('text-primary');
                    if(statusText.innerText === 'Apagado') statusText.innerText = 'Activo';
                    if(statusText.innerText === 'Apagadas') statusText.innerText = 'Encendidas';
                } else {
                    statusText.classList.add('text-on-surface-variant');
                    statusText.classList.remove('text-primary');
                    if(statusText.innerText === 'Activo' || statusText.innerText === 'Regando') statusText.innerText = 'Apagado';
                    if(statusText.innerText === 'Encendidas') statusText.innerText = 'Apagadas';
                }
            });
        });
    </script>
